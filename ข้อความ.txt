        // Update Morphological Sex Calculation (Buikstra & Ubelaker weighted scoring)
        function updateSexEstimation() {
            const sciatic = parseInt(document.getElementById('input-sciatic').value);
            const ventral = parseInt(document.getElementById('input-ventral').value) * 1.66; // Scale 1-3 normalized to 1-5 equivalent
            const nuchal = parseInt(document.getElementById('input-nuchal').value);
            const supraorbital = parseInt(document.getElementById('input-supraorbital').value);
            const mastoid = parseInt(document.getElementById('input-mastoid').value);

            // Update Labels
            document.getElementById('val-sciatic').innerText = `${sciatic} (${sciatic <= 2 ? 'Female' : sciatic >= 4 ? 'Male' : 'Indeterminate'})`;
            document.getElementById('val-ventral').innerText = `${document.getElementById('input-ventral').value} (${ventral <= 2 font='Female' ? 'Female' : 'Male'})`;
            document.getElementById('val-nuchal').innerText = `${nuchal} (${nuchal <= 2 ? 'Gracile' : nuchal >= 4 ? 'Robust' : 'Indeterminate'})`;
            document.getElementById('val-supraorbital').innerText = `${supraorbital} (${supraorbital <= 2 ? 'Sharp/F' : supraorbital >= 4 ? 'Blunt/M' : 'Indeterminate'})`;
            document.getElementById('val-mastoid').innerText = `${mastoid} (${mastoid <= 2 ? 'Small/F' : mastoid >= 4 ? 'Massive/M' : 'Indeterminate'})`;

            // Weighted Sex Score Calculation (Phenice pelvic traits weighted 1.5x)
            const weightedSum = (sciatic * 1.5) + (ventral * 1.5) + nuchal + supraorbital + mastoid;
            const maxWeighted = (5 * 1.5) + (5 * 1.5) + 5 + 5 + 5; // 30
            const maleProbability = Math.round((weightedSum / maxWeighted) * 100);
            const femaleProbability = 100 - maleProbability;

            document.getElementById('male-pct').innerText = `${maleProbability}%`;
            document.getElementById('female-pct').innerText = `${femaleProbability}%`;
            document.getElementById('male-bar').style.width = `${maleProbability}%`;
            document.getElementById('female-bar').style.width = `${femaleProbability}%`;

            // Status Classification Badge
            let statusText = "INDETERMINATE";
            let statusClass = "bg-slate-500/20 text-slate-300 border-slate-500/30";
            if (femaleProbability >= 70) {
                statusText = "PROBABLE FEMALE";
                statusClass = "bg-rose-500/20 text-rose-300 border-rose-500/40";
            } else if (femaleProbability >= 58) {
                statusText = "POSSIBLE FEMALE";
                statusClass = "bg-pink-500/20 text-pink-300 border-pink-500/40";
            } else if (maleProbability >= 70) {
                statusText = "PROBABLE MALE";
                statusClass = "bg-cyan-500/20 text-cyan-300 border-cyan-500/40";
            } else if (maleProbability >= 58) {
                statusText = "POSSIBLE MALE";
                statusClass = "bg-blue-500/20 text-blue-300 border-blue-500/40";
            }

            const badge = document.getElementById('sex-status-badge');
            badge.innerText = statusText;
            badge.className = `px-2 py-0.5 rounded text-[11px] font-mono border ${statusClass}`;

            document.getElementById('summary-sex').innerText = `${statusText} (${Math.max(femaleProbability, maleProbability)}% confidence)`;

            // Update Radar Chart Data
            if (sexRadarChart) {
                sexRadarChart.data.datasets[0].data = [sciatic, ventral, nuchal, supraorbital, mastoid];
                sexRadarChart.update();
            }
        }

        // Stature Calculator (Trotter & Gleser 1952, 1958 Formulas)
        function calculateStature() {
            const femur = parseFloat(document.getElementById('femur-len').value) / 10; // convert to cm
            const tibia = parseFloat(document.getElementById('tibia-len').value) / 10; // convert to cm
            const sex = document.getElementById('stature-sex').value;
            const ancestry = document.getElementById('stature-ancestry').value;

            let heightCm = 0;
            let errMargin = 3.27;

            // Trotter & Gleser Stature Formulas
            if (sex === 'male') {
                if (ancestry === 'asian') {
                    heightCm = (2.15 * femur) + 72.57;
                    errMargin = 3.80;
                } else if (ancestry === 'black') {
                    heightCm = (2.11 * femur) + 70.35;
                    errMargin = 3.94;
                } else { // White
                    heightCm = (2.38 * femur) + 61.41;
                    errMargin = 3.27;
                }
            } else { // Female
                if (ancestry === 'black') {
                    heightCm = (2.28 * femur) + 59.76;
                    errMargin = 3.41;
                } else { // White / Asian female default
                    heightCm = (2.47 * femur) + 54.10;
                    errMargin = 3.72;
                }
            }

            const totalInches = heightCm / 2.54;
            const feet = Math.floor(totalInches / 12);
            const inches = (totalInches % 12).toFixed(1);

            const heightStr = `${heightCm.toFixed(1)} cm`;
            const errStr = `± ${errMargin} cm`;

            document.getElementById('stature-result-text').innerHTML = `${heightStr} <span class="text-xs font-normal text-emerald-400">${errStr}</span>`;
            document.getElementById('stature-ft-text').innerText = `${feet} ft ${inches} in`;
            document.getElementById('summary-stature').innerText = `${heightStr} ${errStr}`;

            const ancestryText = document.getElementById('stature-ancestry').options[document.getElementById('stature-ancestry').selectedIndex].text;
            document.getElementById('summary-ancestry').innerText = ancestryText;
        }
